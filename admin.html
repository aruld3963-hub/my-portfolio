SQL SCHEMA FOR SUPABASE TABLES 
-- 1. Create Achievements Table
CREATE TABLE achievements (
  id TEXT PRIMARY KEY,
  title TEXT NOT NULL,
  issuer TEXT,
  category TEXT,
  date TEXT,
  verificationId TEXT,
  credentialUrl TEXT,
  imageUrl TEXT,
  skills TEXT[],
  description TEXT,
  featured BOOLEAN DEFAULT false,
  is_visible BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Create Projects Table
CREATE TABLE IF NOT EXISTS projects (
  id TEXT PRIMARY KEY,
  title TEXT NOT NULL,
  category TEXT,
  description TEXT,
  skills TEXT[],
  githubUrl TEXT,
  liveUrl TEXT,
  imageUrl TEXT,
  featured BOOLEAN DEFAULT false,
  is_visible BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Create Site Profile & Dynamic CMS Table
CREATE TABLE IF NOT EXISTS site_profile (
  id TEXT PRIMARY KEY DEFAULT 'main_profile',
  full_name TEXT,
  headline_badge TEXT,
  degree_title TEXT,
  college_name TEXT,
  bio_summary TEXT,
  core_skills TEXT[],
  github_url TEXT,
  linkedin_url TEXT,
  email TEXT,
  resume_url TEXT,
  avatar_url TEXT,
  education_journey JSONB DEFAULT '[]'::jsonb,
  is_badge_visible BOOLEAN DEFAULT true,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enable Row Level Security (RLS)
ALTER TABLE site_profile ENABLE ROW LEVEL SECURITY;

-- Allow public read access (SELECT)
CREATE POLICY "Allow public read access on site_profile" 
ON site_profile FOR SELECT USING (true);

-- Allow public/anon upsert (INSERT & UPDATE) for portfolio administration
CREATE POLICY "Allow anon insert on site_profile" 
ON site_profile FOR INSERT WITH CHECK (true);

CREATE POLICY "Allow anon update on site_profile" 
ON site_profile FOR UPDATE USING (true);
